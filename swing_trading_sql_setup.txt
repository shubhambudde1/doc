WITH base AS (
    SELECT *,
        AVG(close) OVER (PARTITION BY symbol ORDER BY date ROWS BETWEEN 49 PRECEDING AND CURRENT ROW) AS ma50,
        AVG(close) OVER (PARTITION BY symbol ORDER BY date ROWS BETWEEN 199 PRECEDING AND CURRENT ROW) AS ma200,
        AVG(volume) OVER (PARTITION BY symbol ORDER BY date ROWS BETWEEN 19 PRECEDING AND CURRENT ROW) AS vol_avg20,
        LAG(close, 22) OVER (PARTITION BY symbol ORDER BY date) AS close_22,
        close - LAG(close) OVER (PARTITION BY symbol ORDER BY date) AS change
    FROM historical_prices
),
rsi_calc AS (
    SELECT *,
        CASE WHEN change > 0 THEN change ELSE 0 END AS gain,
        CASE WHEN change < 0 THEN -change ELSE 0 END AS loss
    FROM base
),
rsi_avg AS (
    SELECT *,
        AVG(gain) OVER (PARTITION BY symbol ORDER BY date ROWS BETWEEN 13 PRECEDING AND CURRENT ROW) AS avg_gain,
        AVG(loss) OVER (PARTITION BY symbol ORDER BY date ROWS BETWEEN 13 PRECEDING AND CURRENT ROW) AS avg_loss
    FROM rsi_calc
),
final AS (
    SELECT *,
        100 - (100 / (1 + (avg_gain / NULLIF(avg_loss, 0)))) AS rsi,
        (close / close_22 - 1) AS monthly_return
    FROM rsi_avg
)

SELECT *,
    -- scoring
    (CASE WHEN close > ma50 THEN 1 ELSE 0 END +
     CASE WHEN ma50 > ma200 THEN 1 ELSE 0 END +
     CASE WHEN rsi BETWEEN 50 AND 70 THEN 1 ELSE 0 END +
     CASE WHEN volume > 1.5 * vol_avg20 THEN 1 ELSE 0 END +
     CASE WHEN monthly_return > 0.05 THEN 1 ELSE 0 END
    ) AS score
FROM final
WHERE date = (SELECT MAX(date) FROM historical_prices)
ORDER BY score DESC
LIMIT 7;
